- A new codec, `source/savegame.c`, converts each saved record to and from its packed little-endian DOS bytes, and makes the same `dfwrite`/`kdfread` calls the DOS game made.

- `-loadverify`/`-saveverify` make loading testable headless.
- An independent Python decoder checks the results against 46 real Atomic saves.

```python
#!/usr/bin/env python3
"""Copy files out of an ISO 9660 CD image without changing it.

  extract_iso_files.py <image> <dest dir> <path on disc>...

The image is a plain .iso (2048-byte sectors) or a raw MODE1/2352 .bin.
Paths are like ATOMINST/DUKE3D.GRP and match case-insensitively; the ";1"
version suffix is ignored. Files are written to <dest dir> under their own
name.
"""
import os
import struct
import sys


class Image:
    def __init__(self, path):
        self.f = open(path, 'rb')
        self.raw = False
        self.f.seek(16 * 2352 + 16 + 1)
        if self.f.read(5) == b'CD001':
            self.raw = True
        elif self.sector(16)[1:6] != b'CD001':
            raise SystemExit('%s: no ISO 9660 volume found' % path)

    def sector(self, lba):
        if self.raw:
            self.f.seek(lba * 2352 + 16)
        else:
            self.f.seek(lba * 2048)
        return self.f.read(2048)

    def extent(self, lba, size):
        out = bytearray()
        while len(out) < size:
            out += self.sector(lba)
            lba += 1
        return bytes(out[:size])

    def entries(self, lba, size):
        data = self.extent(lba, size)
        o = 0
        while o < len(data):
            length = data[o]
            if length == 0:
                o = (o // 2048 + 1) * 2048
                continue
            rec = data[o:o + length]
            name = rec[33:33 + rec[32]]
            if name not in (b'\0', b'\1'):
                yield (name.decode('ascii').split(';')[0].upper(),
                       struct.unpack_from('<I', rec, 2)[0],
                       struct.unpack_from('<I', rec, 10)[0],
                       bool(rec[25] & 2))
            o += length

    def find(self, path):
        pvd = self.sector(16)
        lba = struct.unpack_from('<I', pvd, 156 + 2)[0]
        size = struct.unpack_from('<I', pvd, 156 + 10)[0]
        parts = path.upper().split('/')
        for depth, part in enumerate(parts):
            for name, elba, esize, is_dir in self.entries(lba, size):
                if name == part and is_dir == (depth < len(parts) - 1):
                    lba, size = elba, esize
                    break
            else:
                raise SystemExit('%s: not on the disc' % path)
        return lba, size


def main(argv):
    if len(argv) < 4:
        print(__doc__)
        return 2
    image = Image(argv[1])
    os.makedirs(argv[2], exist_ok=True)
    for path in argv[3:]:
        lba, size = image.find(path)
        dest = os.path.join(argv[2], path.split('/')[-1].upper())
        with open(dest, 'wb') as out:
            out.write(image.extent(lba, size))
        print('%s -> %s (%d bytes)' % (path, dest, size))
    return 0


if __name__ == '__main__':
    sys.exit(main(sys.argv))
```

```python
def uncompress(block):
    """One LZW block (without its 2-byte length) -> its bytes."""
    uncompleng, strtot = struct.unpack_from('<Hh', block, 0)
    if strtot == 0:
        return block[4:4 + uncompleng]
    data = block + b'\0\0\0\0'
    lzwbuf2 = list(range(256)) + [0] * (strtot + 2)
    lzwbuf3 = list(range(256)) + [0] * (strtot + 2)
    out = bytearray()
    currstr, bitcnt, numbits, oneupnumbits = 256, 32, 8, 256
    while True:
        at = bitcnt >> 3
        dat = (int.from_bytes(data[at:at + 4], 'little') >> (bitcnt & 7)) & (oneupnumbits - 1)
        bitcnt += numbits
        half = (oneupnumbits >> 1) - 1
        if (dat & half) > ((currstr - 1) & half):
            dat &= half
            bitcnt -= 1
        lzwbuf3[currstr] = dat
        tail = []
        while dat >= 256:
            tail.append(lzwbuf2[dat])
            dat = lzwbuf3[dat]
        out.append(dat)
        out.extend(reversed(tail))
        lzwbuf2[currstr - 1] = dat
        lzwbuf2[currstr] = dat
        currstr += 1
        if currstr > oneupnumbits:
            numbits += 1
            oneupnumbits <<= 1
        if currstr >= strtot:
            break
    return bytes(out[:uncompleng])


class Save:
    def __init__(self, data):
        self.data = data
        self.blocks = []        # raw blocks, each with its 2-byte length
        self.calls = {}         # name -> the call's bytes, delta-decoded
        self.call_blocks = {}   # name -> (first block index, block count)
        o = 0
        for name, dasizeof, count in CALLS:
            if count is None:
                count = struct.unpack('<h', self.calls['spriteqamount'])[0]
            first = len(self.blocks)
            self.calls[name], o = self._call(o, dasizeof, count)
            self.call_blocks[name] = (first, len(self.blocks) - first)
        if o != len(data):
            raise ValueError('%d bytes left after the last field' % (len(data) - o))

    def _block(self, o):
        if o + 2 > len(self.data):
            raise ValueError('file ends at byte %d' % o)
        leng = struct.unpack_from('<h', self.data, o)[0]
        raw = self.data[o:o + 2 + leng]
        if leng < 4 or len(raw) != 2 + leng:
            raise ValueError('bad block at byte %d' % o)
        self.blocks.append(raw)
        return uncompress(raw[2:]), o + 2 + leng

    def _call(self, o, dasizeof, count):
        # kdfread: each element after the first is a byte-wise delta from
        # the one before; a new block starts when the current one is used up.
        if dasizeof > LZWSIZE:
            count, dasizeof = count * dasizeof, 1
        block, o = self._block(o)
        out = bytearray(block[:dasizeof])
        k = dasizeof
        for _ in range(1, count):
            if k >= len(block):
                block, o = self._block(o)
                k = 0
            prev = out[-dasizeof:]
            out += bytes((prev[j] + block[k + j]) & 255 for j in range(dasizeof))
            k += dasizeof
        return bytes(out), o


def read(path):
    with open(path, 'rb') as f:
        return Save(f.read())


def state(save):
    c = save.calls
    ps0 = c['ps'][:PLAYER_SIZE]
    i = get(PLAYER_OFFS, ps0, 'i')
    spr = c['sprite'][i * SPRITE_SIZE:(i + 1) * SPRITE_SIZE]
    weapon = get(PLAYER_OFFS, ps0, 'curr_weapon')
    return 'SAVESTATE %d %d %d %d %d %d %d %d' % (
        get(SPRITE_OFFS, spr, 'extra'),
        get(PLAYER_OFFS, ps0, 'shield_amount'),
        weapon,
        get(PLAYER_OFFS, ps0, 'ammo_amount', weapon),
        struct.unpack('<i', c['volume'])[0],
        struct.unpack('<i', c['level'])[0],
        get(PLAYER_OFFS, ps0, 'posx'),
        get(PLAYER_OFFS, ps0, 'posy'))


def masked(name, data):
    """Bytes that may differ after a load and an immediate save."""
    if name in ('scriptptrs', 'hitptrs'):
        return b''      # offset bitmaps: DOS fills them, the port writes zeros
    if name == 'ps':
        data = bytearray(data)
        for p in range(MAXPLAYERS):
            o = p * PLAYER_SIZE + PLAYER_OFFS['palette'][0]
            data[o:o + 4] = b'\0\0\0\0'     # a DOS pointer, never restored
        data[PLAYER_OFFS['gm'][0]] = 0      # loadplayer sets ps[0].gm to MODE_GAME
        return bytes(data)
    return data


def compare(a, b):
    bad = 0
    for name, _, _ in CALLS:
        x, y = masked(name, a.calls[name]), masked(name, b.calls[name])
        if x != y:
            n = min(len(x), len(y))
            at = next((j for j in range(n) if x[j] != y[j]), n)
            print('%s differs at byte %d (%d vs %d bytes)' % (name, at, len(x), len(y)))
            bad = 1
    return bad


def fixture(kind, save, out):
    if kind == 'version':
        data = bytearray(save.data)
        if data[0:2] != b'\x08\x00' or data[4:6] != b'\x00\x00':
            raise ValueError('the version block is not stored uncompressed')
        data[6:10] = struct.pack('<i', 103)
    else:
        # The player block as an unpacked 64-bit build wrote it: 920 bytes a
        # player. A stored block (string count 0) needs no compressor.
        first, count = save.call_blocks['ps']
        ps = save.calls['ps']
        old = b''.join(ps[p * PLAYER_SIZE:(p + 1) * PLAYER_SIZE] + b'\0' * 9
                       for p in range(MAXPLAYERS))
        stored = struct.pack('<hHh', 4 + len(old), len(old), 0) + old
        data = (b''.join(save.blocks[:first]) + stored
                + b''.join(save.blocks[first + count:]))
    with open(out, 'wb') as f:
        f.write(data)


def main(argv):
    cmd = argv[1] if len(argv) > 1 else ''
    if cmd == 'check' and len(argv) == 3:
        save = read(argv[2])
        version = struct.unpack('<i', save.calls['version'])[0]
        if version != BYTEVERSION:
            print('version %d, not %d' % (version, BYTEVERSION))
            return 1
        print('ok')
    elif cmd == 'state' and len(argv) == 3:
        print(state(read(argv[2])))
    elif cmd == 'compare' and len(argv) == 4:
        return compare(read(argv[2]), read(argv[3]))
    elif cmd == 'fixture' and len(argv) == 5 and argv[2] in ('oldlayout', 'version'):
        fixture(argv[2], read(argv[3]), argv[4])
    else:
        print(__doc__)
        return 2
    return 0


if __name__ == '__main__':
    try:
        sys.exit(main(sys.argv))
    except ValueError as e:
        print('error: %s' % e)
        sys.exit(1)
```

```c
static int32_t le32(const unsigned char *b)
```

```c
 * every value little-endian, written with the same dfwrite calls (element
 * size and count) the DOS game made, so the LZW blocks come out the same.

#ifndef _INCL_SAVEGAME_H_
#define _INCL_SAVEGAME_H_

#define SG_WALL_SIZE        32
#define SG_SECTOR_SIZE      40
#define SG_SPRITE_SIZE      44

#define SG_PLAYERORIG_SIZE  16
#define SG_WEAPONHIT_SIZE   71
#define SG_ANIMWALL_SIZE    6
#define SG_DOS_MAXTILES     6144    /* actortype[] and actorscrptr[] entries in a save */

/* Converts one record: write != 0 is struct -> bytes, 0 is bytes -> struct.
   Returns the bytes used. */

int32_t sg_wall(unsigned char *b, void *rec, int write);
int32_t sg_sector(unsigned char *b, void *rec, int write);
```

```c
/*
 * Converts saved records to and from the DOS 1.5 layout. See savegame.h.
 */
#include "duke3d.h"
#include "savegame.h"

#define SG_LZWSIZE 16384    /* LZWSIZE in build/cache1d.c */

static unsigned char sgbuf[MAXSPRITES*SG_WEAPONHIT_SIZE];   /* the largest block */

static void put16(unsigned char *b, int32_t v) { b[0] = (unsigned char)v; b[1] = (unsigned char)(v>>8); }
static void put32(unsigned char *b, int32_t v) { put16(b,v); put16(b+2,v>>16); }
static int32_t get16(const unsigned char *b) { return (short)(b[0]|(b[1]<<8)); }
static int32_t get32(const unsigned char *b)
{
    return (int32_t)((uint32_t)b[0]|((uint32_t)b[1]<<8)|((uint32_t)b[2]<<16)|((uint32_t)b[3]<<24));
}

/* One field each, in declaration order. Uses b, write, o and k. */
#define F8(x)    do { if (write) b[o] = (unsigned char)(x); else (x) = b[o]; o += 1; } while (0)
#define F16(x)   do { if (write) put16(b+o,(x)); else (x) = get16(b+o); o += 2; } while (0)
#define F32(x)   do { if (write) put32(b+o,(x)); else (x) = get32(b+o); o += 4; } while (0)
#define A8(a,n)  do { for (k = 0; k < (n); k++) F8((a)[k]); } while (0)
#define A16(a,n) do { for (k = 0; k < (n); k++) F16((a)[k]); } while (0)
#define A32(a,n) do { for (k = 0; k < (n); k++) F32((a)[k]); } while (0)

int32_t sg_wall(unsigned char *b, void *rec, int write)
{

    F32(w->x); F32(w->y);

}

    kdfread(sgbuf,dasizeof,count,fil);
    for (i = 0; i < n; i++) o += fn(sgbuf+o,(char *)recs+i*stride,0);

/* Mirrors dfwrite's chunking in build/cache1d.c. */
int32_t sg_dfblocks(int32_t dasizeof, int32_t count)
{
    int32_t i, k, n = 0;

    if (dasizeof > SG_LZWSIZE) { count *= dasizeof; dasizeof = 1; }
    k = dasizeof;
    if (k > SG_LZWSIZE-dasizeof) { n++; k = 0; }
    for (i = 1; i < count; i++)
    {
        k += dasizeof;
        if (k > SG_LZWSIZE-dasizeof) { n++; k = 0; }
    }
    if (k > 0) n++;
    return n;
}

int32_t sg_playerblocksize(int32_t fil)
{
    /* The dfwrite calls between the version block and ps[]. */
    static const int32_t calls[][2] =
    {
        {4,1}, {19,1}, {4,1}, {4,1}, {4,1}, {160,100},
        {2,1}, {SG_WALL_SIZE,MAXWALLS}, {2,1}, {SG_SECTOR_SIZE,MAXSECTORS},
        {SG_SPRITE_SIZE,MAXSPRITES},
        {2,MAXSECTORS+1}, {2,MAXSPRITES}, {2,MAXSPRITES},
        {2,MAXSTATUS+1}, {2,MAXSPRITES}, {2,MAXSPRITES},
        {2,1}, {12,MAXCYCLERS}
    };
    int32_t start = klseek(fil,0,SEEK_CUR), i, n = 0, size = -1;
    short leng, uncompleng;

    for (i = 0; i < (int32_t)(sizeof(calls)/sizeof(calls[0])); i++)
        n += sg_dfblocks(calls[i][0],calls[i][1]);
    for (i = 0; i < n; i++)
    {
        if (!kread16(fil,&leng)) break;
        klseek(fil,leng,SEEK_CUR);
    }
    if (i == n && kread16(fil,&leng) && kread16(fil,&uncompleng))
        size = (unsigned short)uncompleng;
    klseek(fil,start,SEEK_SET);
    return size;
}
```

The widths (8, 16 or 32 bits) are the DOS on-disk widths. They're fixed by the file, whatever type the variable has. If a `sg_read16s`/`sg_read32s` call gets a pointer of the wrong element type (an `int32_t` array passed as `short *`, say), the compiler warns. Stop and report it rather than casting it away: it would mean the DOS layout and our types disagree.
